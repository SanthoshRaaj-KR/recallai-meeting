```tsx
import { CheckCircle2, Loader2, Circle } from "lucide-react";
import type { PipelineStage } from "@/types";
import { cn } from "@/lib/utils";

const STAGE_SEQUENCE: PipelineStage[] = [
  "fact_extraction",
  "rag_retrieval",
  "drafting",
  "verification",
];

const STAGE_LABELS: Record<PipelineStage, string> = {
  fact_extraction: "Extracting facts",
  rag_retrieval:   "Retrieving Confluence pages",
  drafting:        "Drafting proposals",
  verification:    "Verifying changes",
};

type Status = "completed" | "active" | "pending";

function getStatus(
  stage: PipelineStage,
  currentStage: PipelineStage | null,
  completedStages: PipelineStage[],
): Status {
  if (completedStages.includes(stage)) return "completed";
  if (stage === currentStage) return "active";
  return "pending";
}

interface StageIndicatorProps {
  currentStage: PipelineStage | null;
  completedStages: PipelineStage[];
  isLive: boolean;
  isComplete: boolean;
  proposalCount?: number;
}

export const StageIndicator = ({
  currentStage,
  completedStages,
  isLive,
  isComplete,
  proposalCount = 0,
}: StageIndicatorProps) => {
  if (isComplete) {
    return (
      <div
        className="bg-card rounded-2xl border border-border shadow-soft p-5 flex items-center gap-3"
        role="status"
        aria-label="Pipeline progress"
      >
        <CheckCircle2 className="size-5 text-brand-emerald shrink-0" />
        <span className="text-sm font-semibold">
          Pipeline complete — {proposalCount} proposal{proposalCount === 1 ? "" : "s"} ready
        </span>
      </div>
    );
  }

  return (
    <div
      className="bg-card rounded-2xl border border-border shadow-soft p-5 relative"
      aria-label="Pipeline progress"
    >
      {isLive && (
        <div className="absolute right-5 top-5 flex items-center gap-2">
          <div className="size-2 rounded-full bg-brand-blue animate-pulse" aria-hidden="true" />
          <span className="text-xs text-muted-foreground">Running...</span>
        </div>
      )}
      <ol className="flex flex-col sm:flex-row items-start sm:items-center gap-4">
        {STAGE_SEQUENCE.map((stage, idx) => {
          const status = getStatus(stage, currentStage, completedStages);
          const isLast = idx === STAGE_SEQUENCE.length - 1;
          return (
            <li key={stage} role="listitem" className="flex items-center gap-3 sm:flex-1 min-w-0">
              <div className="flex flex-col sm:flex-row items-center gap-2 sm:gap-3 min-w-0">
                <div
                  className={cn(
                    "size-8 rounded-full flex items-center justify-center shrink-0",
                    status === "completed" && "bg-brand-emerald-soft",
                    status === "active" && "bg-brand-blue-soft ring-2 ring-brand-blue ring-offset-2",
                    status === "pending" && "bg-secondary border border-border",
                  )}
                >
                  {status === "completed" && <CheckCircle2 className="size-4 text-brand-emerald" />}
                  {status === "active" && <Loader2 className="size-4 text-brand-blue animate-spin" />}
                  {status === "pending" && <Circle className="size-4 text-muted-foreground" />}
                </div>
                <span
                  className={cn(
                    "text-xs font-semibold whitespace-nowrap",
                    status === "completed" && "text-brand-emerald",
                    status === "active" && "text-brand-blue",
                    status === "pending" && "text-muted-foreground font-normal",
                  )}
                >
                  {STAGE_LABELS[stage]}
                </span>
              </div>
              {!isLast && (
                <div
                  className={cn(
                    "hidden sm:block flex-1 h-0.5",
                    status === "completed" && "bg-brand-emerald/30",
                    status === "active" && "bg-brand-blue/40 animate-pulse",
                    status === "pending" && "bg-border",
                  )}
                  aria-hidden="true"
                />
              )}
            </li>
          );
        })}
      </ol>
    </div>
  );
};

export default StageIndicator;
```

```tsx
import { FileText } from "lucide-react";
import type { ChangeItem } from "@/types";
import { ProposalCard } from "./ProposalCard";

const CONFIDENCE_ORDER: Record<"high" | "medium" | "low", number> = {
  high: 0,
  medium: 1,
  low: 2,
};

interface ProposalCardGroupProps {
  pageTitle: string;
  cards: Array<ChangeItem & { id: string }>;
  sessionId: string;
  onAccept: (id: string) => Promise<void>;
  onReject: (id: string) => void;
  cardStatuses: Record<string, "pending" | "applying" | "accepted" | "rejected">;
}

export const ProposalCardGroup = ({
  pageTitle,
  cards,
  sessionId,
  onAccept,
  onReject,
  cardStatuses,
}: ProposalCardGroupProps) => {
  const sorted = [...cards].sort((a, b) => {
    const ac = (a.confidence ?? "low") as "high" | "medium" | "low";
    const bc = (b.confidence ?? "low") as "high" | "medium" | "low";
    return CONFIDENCE_ORDER[ac] - CONFIDENCE_ORDER[bc];
  });

  return (
    <section className="space-y-3 mb-8">
      <div className="flex items-center gap-2 mb-3">
        <FileText className="size-4 text-muted-foreground" />
        <h2 className="text-lg font-semibold">{pageTitle}</h2>
        <span className="text-xs text-muted-foreground ml-1">
          ({sorted.length} change{sorted.length === 1 ? "" : "s"})
        </span>
      </div>
      {sorted.map((card) => (
        <ProposalCard
          key={card.id}
          card={card}
          sessionId={sessionId}
          status={cardStatuses[card.id] ?? "pending"}
          onAccept={() => onAccept(card.id)}
          onReject={() => onReject(card.id)}
        />
      ))}
    </section>
  );
};

export default ProposalCardGroup;
```

          <p className="text-sm text-foreground leading-relaxed">{card.verifier_note}</p>
        </div>
      )}

      {/* Region 8: Delete confirmation checkbox */}

        <div className="flex items-start gap-3 mb-4 p-3 rounded-lg border border-destructive/20 bg-destructive/5">
          <Checkbox
            id={`confirm-delete-${card.id}`}
            checked={deleteConfirmed}
            onCheckedChange={(checked) => setDeleteConfirmed(checked === true)}
          />
          <label
            htmlFor={`confirm-delete-${card.id}`}
            className="text-sm text-foreground cursor-pointer leading-relaxed"
          >
            I understand this deletion cannot be undone
          </label>
        </div>
      )}

      {/* Region 9: Action row */}
      {isPending || isApplying ? (
        <div className="flex items-center gap-3 mt-2">
          <Button
            variant="outline"
            size="sm"
            className="rounded-full border-destructive text-destructive hover:bg-destructive/10"
            onClick={onReject}
            disabled={!isPending}
            aria-label="Reject change"
          >
            Reject
          </Button>
          <Button
            size="sm"
            className="rounded-full"
            onClick={handleAcceptClick}
            disabled={acceptDisabled}
            aria-label="Accept change"
          >
            {isApplying && <Loader2 className="size-3.5 animate-spin mr-1" />}
            {isApplying ? "Applying..." : "Accept"}
          </Button>
        </div>
      ) : null}
    </article>
  );
};

import { useEffect, useMemo, useRef, useState } from "react";
import { useParams, useSearchParams, Link } from "react-router-dom";
import { toast } from "sonner";
import { AlertCircle, FileText } from "lucide-react";
import { Navbar } from "@/components/Navbar";
import { Button } from "@/components/ui/button";
import { StageIndicator } from "@/components/StageIndicator";
import { ProposalCardGroup } from "@/components/ProposalCardGroup";
import { ProposalCardSkeleton } from "@/components/ProposalCardSkeleton";
import { openPipelineStream, executeProposal } from "@/lib/api";
import type { ChangeItem, PipelineStage } from "@/types";

type CardStatus = "pending" | "applying" | "accepted" | "rejected";

const PipelinePage = () => {
  const { jobId = "" } = useParams<{ jobId: string }>();
  const [params] = useSearchParams();
  const sessionId = params.get("session") ?? "";

  const [currentStage, setCurrentStage] = useState<PipelineStage | null>(null);
  const [completedStages, setCompletedStages] = useState<PipelineStage[]>([]);
  const [cards, setCards] = useState<Array<ChangeItem & { id: string }>>([]);
  const [cardStatuses, setCardStatuses] = useState<Record<string, CardStatus>>({});
  const [streamComplete, setStreamComplete] = useState(false);
  const [proposalCount, setProposalCount] = useState(0);
  const [error, setError] = useState<string | null>(null);
  const [disconnected, setDisconnected] = useState(false);

  const esRef = useRef<EventSource | null>(null);

  useEffect(() => {
    let mounted = true;

    openPipelineStream(jobId, {
      onStageStart: (stage) => {
        if (!mounted) return;
        setCurrentStage((prev) => {
          if (prev && prev !== stage) {
            setCompletedStages((done) => (done.includes(prev) ? done : [...done, prev]));
          }
          return stage as PipelineStage;
        });
      },
      onProposalReady: (card) => {
        if (!mounted) return;
        setCards((prev) => {
          if (prev.some((c) => c.id === card.id)) return prev;
          return [...prev, card];
        });
        setCardStatuses((prev) => ({ ...prev, [card.id]: "pending" }));
      },
      onPipelineComplete: (count) => {
        if (!mounted) return;
        setProposalCount(count);
        setStreamComplete(true);
        setCurrentStage((prev) => {
          if (prev) {
            setCompletedStages((done) => (done.includes(prev) ? done : [...done, prev]));
          }
          return null;
        });
        if (sessionId) {
          localStorage.removeItem(`activeJob:${sessionId}`);
        }
      },
      onPipelineError: (detail) => {
        if (!mounted) return;
        setError(detail);
        if (sessionId) {
          localStorage.removeItem(`activeJob:${sessionId}`);
        }
      },
    })
      .then((es) => {
        if (!mounted) {
          es.close();
          return;
        }
        esRef.current = es;
        es.onerror = () => {
          // Only mark disconnected if stream is still expected to continue.
          if (!streamComplete && !error) {
            setDisconnected(true);
          }
        };
      })
      .catch((err) => {
        if (!mounted) return;
        setError(err instanceof Error ? err.message : "Could not open stream.");
      });

    return () => {
      mounted = false;
      esRef.current?.close();
      esRef.current = null;
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [jobId]);

  const groupedCards = useMemo(() => {
    const groups = new Map<string, Array<ChangeItem & { id: string }>>();
    for (const card of cards) {
      const key = card.page_title ?? "(Untitled page)";
      const arr = groups.get(key);
      if (arr) arr.push(card);
      else groups.set(key, [card]);
    }
    return Array.from(groups.entries()).map(([pageTitle, items]) => ({ pageTitle, items }));
  }, [cards]);

  const handleAccept = async (id: string) => {
    if (!sessionId) {
      toast.error("Session ID missing — cannot apply change.");
      return;
    }
    setCardStatuses((prev) => ({ ...prev, [id]: "applying" }));
    try {
      await executeProposal(sessionId, id);
      setCardStatuses((prev) => ({ ...prev, [id]: "accepted" }));
      toast.success("Change accepted and applied to Confluence.");
    } catch (err) {
      setCardStatuses((prev) => ({ ...prev, [id]: "pending" }));
      toast.error(
        err instanceof Error
          ? "Failed to apply change. The page may have been modified. Try again."
          : "Failed to apply change. Try again.",
      );
    }
  };

  const handleReject = (id: string) => {
    setCardStatuses((prev) => ({ ...prev, [id]: "rejected" }));
  };

  const backHref = sessionId
    ? `/meeting/summary?session=${encodeURIComponent(sessionId)}`
    : "/meeting/summary";

  const showSkeletons = !streamComplete && !error && cards.length === 0;

  return (
    <div className="min-h-dvh bg-aura">
      <Navbar />
      <main className="px-6 md:px-8 max-w-6xl mx-auto pt-8 pb-24">
        <Link
          to={backHref}
          className="text-sm text-muted-foreground hover:text-foreground inline-block mb-4"
        >
          ← Back to summary
        </Link>

        <header className="mb-6">
          <h1 className="text-3xl font-semibold leading-none">Review Proposed Changes</h1>
          <p className="text-sm text-muted-foreground mt-2">
            Accept or reject each proposed Confluence update
          </p>
        </header>

        {/* Error banner replaces StageIndicator on pipeline_error */}
        {error ? (
          <div
            role="alert"
            className="rounded-2xl border border-destructive/20 bg-destructive/5 p-6 flex gap-4 mb-6"
          >
            <AlertCircle className="size-5 text-destructive shrink-0 mt-0.5" />
            <div>
              <p className="text-sm font-semibold text-destructive">Pipeline error</p>
              <p className="text-sm text-muted-foreground mt-1">
                The pipeline encountered an error and could not complete.{" "}
                {error ? `${error}` : "Please return to the meeting summary and try again."}
              </p>
              <Link
                to={backHref}
                className="text-sm font-medium text-brand-blue hover:underline mt-2 inline-block"
              >
                ← Back to meeting summary
              </Link>
            </div>
          </div>
        ) : (
          <div className="sticky top-4 z-10 mb-6">
            <StageIndicator
              currentStage={currentStage}
              completedStages={completedStages}
              isLive={!streamComplete}
              isComplete={streamComplete}
              proposalCount={proposalCount}
            />
            {disconnected && !streamComplete && (
              <p
                role="status"
                className="mt-2 text-xs text-muted-foreground italic"
              >
                Connection lost — progress updates paused. Cards already received remain reviewable.
              </p>
            )}
          </div>
        )}

        {/* Cards or skeletons or empty state */}
        {showSkeletons && (
          <div className="space-y-3">
            <ProposalCardSkeleton />
            <ProposalCardSkeleton />
          </div>
        )}

        {groupedCards.map(({ pageTitle, items }) => (
          <ProposalCardGroup
            key={pageTitle}
            pageTitle={pageTitle}
            cards={items}
            sessionId={sessionId}
            onAccept={handleAccept}
            onReject={handleReject}
            cardStatuses={cardStatuses}
          />
        ))}

        {/* Empty state — only after pipeline_complete with zero cards */}
        {streamComplete && cards.length === 0 && (
          <div className="rounded-2xl border border-dashed border-border p-12 text-center">
            <div className="mx-auto mb-4 flex size-12 items-center justify-center rounded-full bg-secondary">
              <FileText className="size-6 text-muted-foreground" />
            </div>
            <p className="text-sm font-semibold text-foreground mb-1">No changes proposed</p>
            <p className="text-sm text-muted-foreground max-w-sm mx-auto">
              Jarvis did not find any Confluence pages that need updating based on this meeting.
            </p>
            <Link to={backHref}>
              <Button variant="outline" className="rounded-full mt-6">
                ← Back to summary
              </Button>
            </Link>
          </div>
        )}
      </main>
    </div>
  );
};

export default PipelinePage;
