```typescript
import type { ChangeItem } from "@/types";

export function startPipeline(sessionId: string): Promise<{ job_id: string; status: string }> {
  return request<{ job_id: string; status: string }>("/review/pipeline/start", {
    method: "POST",
    body: JSON.stringify({ session_id: sessionId }),
  });
}

export interface SsePipelineHandlers {
  onStageStart: (stage: string) => void;
  onProposalReady: (card: ChangeItem & { id: string }) => void;
  onPipelineComplete: (proposalCount: number) => void;
  onPipelineError: (detail: string) => void;
}

export async function openPipelineStream(
  jobId: string,
  handlers: SsePipelineHandlers,
): Promise<EventSource> {
  let token = "";
  if (isSupabaseConfigured) {
    const { data } = await supabase.auth.getSession();
    token = data.session?.access_token ?? "";
  }
  const url = `${BASE}/review/pipeline/${encodeURIComponent(jobId)}/stream?token=${encodeURIComponent(token)}`;
  const es = new EventSource(url);

  es.addEventListener("stage_start", (e) => {
    try {
      const data = JSON.parse((e as MessageEvent).data) as { stage: string };
      handlers.onStageStart(data.stage);
    } catch {
      // Ignore malformed events
    }
  });

  es.addEventListener("proposal_ready", (e) => {
    try {
      const data = JSON.parse((e as MessageEvent).data) as ChangeItem & { id: string };
      handlers.onProposalReady(data);
    } catch {
      // Ignore malformed events
    }
  });

  es.addEventListener("pipeline_complete", (e) => {
    try {
      const data = JSON.parse((e as MessageEvent).data) as { proposal_count: number };
      handlers.onPipelineComplete(data.proposal_count);
    } catch {
      handlers.onPipelineComplete(0);
    }
    es.close();
  });

  es.addEventListener("pipeline_error", (e) => {
    try {
      const data = JSON.parse((e as MessageEvent).data) as { detail?: string };
      handlers.onPipelineError(data.detail ?? "Pipeline failed.");
    } catch {
      handlers.onPipelineError("Pipeline failed.");
    }
    es.close();
  });

  return es;
}
```

  if (!resolvedSessionId) {
    toast.error("Session is not ready yet.");
    return;
  }
  setGeneratingPipeline(true);
  try {
    const { job_id } = await startPipeline(resolvedSessionId);
    localStorage.setItem(`activeJob:${resolvedSessionId}`, job_id);
    setActiveJobId(job_id);

  } finally {
    setGeneratingPipeline(false);
  }
};

const dismissPipelineBanner = () => {

  }
  setActiveJobId(null);
};
