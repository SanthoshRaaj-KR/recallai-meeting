    ```python
    class ActionItem(BaseModel):
        owner: str
        task: str
        due: str | None = None

    class MeetingRecord(BaseModel):
        meeting_id: str
        channel_id: str
        channel_name: str
        start_ts: int                          # Unix epoch integer — NEVER datetime
        end_ts: int | None = None
        duration_seconds: int | None = None
        participants: list[str] = Field(default_factory=list)
        summary_text: str
        topics_covered: list[str] = Field(default_factory=list)
        action_items: list[ActionItem] = Field(default_factory=list)
        decisions: list[str] = Field(default_factory=list)
        series_name: str | None = None
        recurrence_pattern: str | None = None
        status: str = "complete"               # "complete" or "partial"
        raw_transcript_chars: int | None = None
        summarized_at: int | None = None       # Unix epoch integer
    ```

    Key design notes:
    - `start_ts` is `int`, not `datetime` — per locked decision "All timestamps stored as Unix epoch integers"
    - `status` field defaults to "complete"; "partial" used for mid-meeting summaries (Pitfall 16 prevention)
    - No nested JSON blobs — all fields are flat scalars or lists of scalars (Pitfall 7 prevention)

    Create `storage/metadata_store.py` with class `MetadataStore`:

    ```python
    class MetadataStore:
        def __init__(self, base_dir: str = "./meetings"):
            self._base_dir = Path(base_dir)

        async def write(self, record: MeetingRecord) -> Path:
            """Write meeting record as JSON file. Returns the file path."""
            dir_path = self._base_dir / record.channel_id
            dir_path.mkdir(parents=True, exist_ok=True)
            file_path = dir_path / f"{record.meeting_id}.json"
            async with aiofiles.open(file_path, "w") as f:
                await f.write(record.model_dump_json(indent=2))
            return file_path

        async def read(self, channel_id: str, meeting_id: str) -> MeetingRecord:
            """Read meeting record from JSON file. Raises FileNotFoundError if not found."""
            file_path = self._base_dir / channel_id / f"{meeting_id}.json"
            async with aiofiles.open(file_path, "r") as f:
                data = await f.read()
            return MeetingRecord.model_validate_json(data)

        async def list_by_channel(self, channel_id: str) -> list[str]:
            """List all meeting_ids for a channel. Returns empty list if channel dir doesn't exist."""
            dir_path = self._base_dir / channel_id
            if not dir_path.exists():
                return []
            return [p.stem for p in dir_path.glob("*.json")]
    ```

    Use `aiofiles` for all file I/O (per research/STACK.md — blocking I/O in async handlers stalls the event loop).
    Use `pathlib.Path` for all path operations.
    Follow existing conventions: docstrings on all public methods, snake_case, logger with emoji prefixes.
