    ```python
    import logging
    from typing import Optional
    from openai import OpenAI
    from pinecone import Pinecone, ServerlessSpec
    from storage.models import MeetingRecord

    logger = logging.getLogger(__name__)

    DENSE_MODEL = "text-embedding-3-small"
    DENSE_DIMENSION = 1536
    SPARSE_MODEL = "pinecone-sparse-english-v0"
    INDEX_METRIC = "dotproduct"

    class PineconeClient:
        def __init__(
            self,
            api_key: str,
            index_name: str = "meeting-memory",
            cloud: str = "aws",
            region: str = "us-east-1",
            openai_api_key: str | None = None,
        ):
            self._pc = Pinecone(api_key=api_key)
            self._openai = OpenAI(api_key=openai_api_key) if openai_api_key else OpenAI()
            self._index_name = index_name
            self._cloud = cloud
            self._region = region
            self._index = None  # lazy-loaded

        def ensure_index_exists(self) -> None:
            """Create the Pinecone index if it does not exist. Uses dotproduct metric (required for hybrid search)."""
            existing = [idx.name for idx in self._pc.list_indexes()]
            if self._index_name in existing:
                logger.info(f"Pinecone index '{self._index_name}' already exists")
                return
            self._pc.create_index(
                name=self._index_name,
                dimension=DENSE_DIMENSION,
                metric=INDEX_METRIC,
                vector_type="dense",
                spec=ServerlessSpec(cloud=self._cloud, region=self._region),
            )
            logger.info(f"Created Pinecone index '{self._index_name}' (metric={INDEX_METRIC}, dim={DENSE_DIMENSION})")

        def _get_index(self):
            """Lazy-load the index handle."""
            if self._index is None:
                self._index = self._pc.Index(self._index_name)
            return self._index

        def _embed_dense(self, text: str) -> list[float]:
            """Generate dense embedding via OpenAI text-embedding-3-small."""
            response = self._openai.embeddings.create(model=DENSE_MODEL, input=[text])
            return response.data[0].embedding

        def _embed_sparse(self, text: str, input_type: str = "passage") -> dict:
            """Generate sparse embedding via Pinecone inference (pinecone-sparse-english-v0).
            Returns {"indices": [...], "values": [...]}."""
            response = self._pc.inference.embed(
                model=SPARSE_MODEL,
                inputs=[text],
                parameters={"input_type": input_type},
            )
            embedding = response.data[0]
            return {
                "indices": [entry.index for entry in embedding.sparse_values],
                "values": [entry.value for entry in embedding.sparse_values],
            }

        def upsert_meeting(self, record: MeetingRecord) -> None:
            """Upsert a meeting record as a hybrid vector (dense + sparse) with flat metadata."""
            dense = self._embed_dense(record.summary_text)
            sparse = self._embed_sparse(record.summary_text, input_type="passage")
            metadata = {
                "channel_id": record.channel_id,
                "channel_name": record.channel_name,
                "start_ts": record.start_ts,
                "end_ts": record.end_ts,
                "duration_seconds": record.duration_seconds,
                "participants": record.participants,
                "topics_covered": record.topics_covered,
                "decisions": record.decisions,
                "series_name": record.series_name,
                "summary_text": record.summary_text,
                "status": record.status,
            }
            self._get_index().upsert(vectors=[{
                "id": record.meeting_id,
                "values": dense,
                "sparse_values": sparse,
                "metadata": metadata,
            }])
            logger.info(f"Upserted meeting {record.meeting_id} to Pinecone")

        def query(
            self,
            query_text: str,
            channel_id: str | None = None,
            start_ts: int | None = None,
            end_ts: int | None = None,
            top_k: int = 10,
        ) -> list[dict]:
            """Hybrid query with optional channel_id and date range filters.
            Returns list of match dicts with id, score, metadata."""
            dense = self._embed_dense(query_text)
            sparse = self._embed_sparse(query_text, input_type="query")

            # Build metadata filter
            filter_conditions = {}
            if channel_id:
                filter_conditions["channel_id"] = {"$eq": channel_id}
            if start_ts is not None:
                filter_conditions.setdefault("start_ts", {})["$gte"] = start_ts
            if end_ts is not None:
                filter_conditions.setdefault("start_ts", {})["$lte"] = end_ts

            query_kwargs = {
                "vector": dense,
                "sparse_vector": sparse,
                "top_k": top_k,
                "include_metadata": True,
            }
            if filter_conditions:
                query_kwargs["filter"] = filter_conditions

            response = self._get_index().query(**query_kwargs)
            return [
                {"id": m.id, "score": m.score, "metadata": m.metadata}
                for m in response.matches
            ]
    ```

    CRITICAL design notes:
    - `metric="dotproduct"` is hardcoded in `ensure_index_exists()` — per locked decision and Pitfall 1 (wrong metric requires full re-ingestion)
    - Sparse vectors use `pinecone-sparse-english-v0` via `pc.inference.embed()` — per INFRA-04 requirement (no local corpus fitting)
    - Metadata is flat scalars/lists only — per Pitfall 7 (nested JSON prevents server-side filtering)
    - `_embed_sparse` uses `input_type="passage"` for upsert and `input_type="query"` for query — per Pinecone docs
    - `start_ts` filter uses `$gte`/`$lte` on integer Unix epoch — per locked decision

    NOTE on `_embed_sparse` return format: The Pinecone SDK v8 `inference.embed()` returns `EmbeddingsList`. Each embedding has a `sparse_values` attribute which is a list of `SparseValue` objects with `.index` and `.value` properties. The executor MUST check the actual SDK response structure at implementation time (the attribute names may be `indices`/`values` directly on the embedding object rather than on a `.sparse_values` sub-object). Read the Pinecone SDK v8 source or test with a real call to confirm the exact shape. If the structure differs, adapt accordingly but keep the output as `{"indices": [...], "values": [...]}`.
